    <script>
        let employees = JSON.parse(localStorage.getItem('pc_employees')) || [
            { id: 1, name: 'Иван Петров', rate: 1500 },
            { id: 2, name: 'Анна Смирнова', rate: 1600 },
            { id: 3, name: 'Сергей Волков', rate: 1400 }
        ];
        const shifts = JSON.parse(localStorage.getItem('pc_shifts_v4')) || {};

        function saveEmployees() { localStorage.setItem('pc_employees', JSON.stringify(employees)); }
        function saveShifts() { localStorage.setItem('pc_shifts_v4', JSON.stringify(shifts)); }

        function showTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));
            document.getElementById(tabName).classList.add('active');
            event.target.classList.add('active');
            if (tabName === 'calendar') renderCalendar();
            else if (tabName === 'employees') renderEmployeeList();
            else if (tabName === 'stats') recalcStats();
        }

        function renderCalendar() {
            const grid = document.getElementById('calendarGrid');
            grid.innerHTML = '';
            const monthVal = document.getElementById('calendarMonth').value || new Date().toISOString().slice(0,7);
            document.getElementById('calendarMonth').value = monthVal;
            const [year, monthNum] = monthVal.split('-').map(Number);
            const firstDay = new Date(year, monthNum - 1, 1);
            const lastDay = new Date(year, monthNum, 0);
            const weekdays = ['Пн','Вт','Ср','Чт','Пт','Сб','Вс'];
            weekdays.forEach(day => {
                const h = document.createElement('div'); h.className = 'day-card day-header'; h.textContent = day; grid.appendChild(h);
            });
            const startOffset = firstDay.getDay() === 0 ? 6 : firstDay.getDay() - 1;
            for (let i=0; i<startOffset; i++) grid.appendChild(document.createElement('div'));
            for (let day=1; day<=lastDay.getDate(); day++) {
                const dateStr = `${year}-${String(monthNum).padStart(2,'0')}-${String(day).padStart(2,'0')}`;
                const card = document.createElement('div'); card.className = 'day-card'; card.dataset.date = dateStr;
                if (shifts[dateStr] && shifts[dateStr].length > 0) card.classList.add('has-shift');
                card.innerHTML = `<div class="day-number">${day}</div><div class="shift-list"></div>`;
                const shiftList = card.querySelector('.shift-list');
                if (shifts[dateStr]) {
                    shifts[dateStr].forEach((s, index) => {
                        const div = document.createElement('div'); div.className = 'shift-item';
                        const span = document.createElement('span'); span.textContent = `${s.employee} (${s.break}ч)`;
                        const btn = document.createElement('button'); btn.textContent = '✕'; btn.className = 'danger';
                        btn.onclick = (e) => { e.stopPropagation(); if(confirm('Удалить?')) { shifts[dateStr].splice(index,1); if(shifts[dateStr].length===0){ delete shifts[dateStr]; card.classList.remove('has-shift'); } saveShifts(); renderCalendar(); recalcStats(); }};
                        div.appendChild(span); div.appendChild(btn); shiftList.appendChild(div);
                    });
                }
                card.onclick = () => openShiftModalForDate(dateStr);
                grid.appendChild(card);
            }
        }
        function openShiftModalForDate(date) { document.getElementById('modalDate').value = date; initSelect(); document.getElementById('shiftModal').classList.add('active'); }
        function closeShiftModal() { document.getElementById('shiftModal').classList.remove('active'); }
        function saveShiftFromModal() {
            const date = document.getElementById('modalDate').value;
            const employee = document.getElementById('modalEmployee').value;
            const breakHours = parseFloat(document.getElementById('modalBreak').value);
            if (!date || !employee) { alert('Заполните поля'); return; }
            if (!shifts[date]) shifts[date] = []; shifts[date].push({ employee, break: breakHours });
            saveShifts(); closeShiftModal(); renderCalendar(); recalcStats();
        }

        function addEmployee() {
            const name = document.getElementById('newEmployeeName').value.trim();
            const rate = parseFloat(document.getElementById('newEmployeeRate').value);
            if (!name || isNaN(rate)) { alert('Введите данные'); return; }
            employees.push({ id: Date.now(), name, rate }); saveEmployees();
            document.getElementById('newEmployeeName').value = ''; document.getElementById('newEmployeeRate').value = '';
            renderEmployeeList(); initSelect(); populateFilterSelect();
        }
        function deleteEmployee(id) { if(!confirm('Удалить?')) return; employees = employees.filter(e => e.id !== id); saveEmployees(); renderEmployeeList(); initSelect(); populateFilterSelect(); recalcStats(); }
        function editEmployee(id) { const emp = employees.find(e => e.id === id); if (!emp) return; const nr = parseFloat(prompt('Ставка:', emp.rate)); if(!isNaN(nr)) { emp.rate = nr; saveEmployees(); renderEmployeeList(); recalcStats(); } }
        function renderEmployeeList() {
            const list = document.getElementById('employeeList'); list.innerHTML = '';
            employees.forEach(e => {
                const li = document.createElement('li'); li.className = 'employee-item';
                li.innerHTML = `<span><b>${e.name}</b> — ${e.rate} ₽</span><div><button class="warning" onclick="editEmployee(${e.id})">✎</button><button class="danger" onclick="deleteEmployee(${e.id})">✕</button></div>`;
                list.appendChild(li);
            });
        }
        function initSelect() { const s = document.getElementById('modalEmployee'); s.innerHTML = ''; employees.forEach(e => { const o=document.createElement('option'); o.value=e.name; o.textContent=e.name; s.appendChild(o); }); }
        function populateFilterSelect() { const s = document.getElementById('filterEmployee'); s.length=1; employees.forEach(e => { const o=document.createElement('option'); o.value=e.name; o.textContent=e.name; s.appendChild(o); }); }

        function recalcStats() {
            const tbody = document.getElementById('statsBody'); tbody.innerHTML = '';
            const selected = document.getElementById('filterEmployee').value;
            const daysWorked = {}; Object.keys(shifts).forEach(date => { const list = selected==='all' ? shifts[date] : shifts[date].filter(s=>s.employee===selected); list.forEach(s => { daysWorked[s.employee] = (daysWorked[s.employee]||0)+1; }); });
            const emps = selected==='all' ? employees : employees.filter(e=>e.name===selected);
            let total=0; emps.forEach(e => { const days=daysWorked[e.name]||0; const sum=days*e.rate; total+=sum; const tr=document.createElement('tr'); tr.innerHTML=`<td class="left">${e.name}</td><td>${e.rate}</td><td>${days}</td><td>${sum}</td>`; tbody.appendChild(tr); });
            document.getElementById('totalSummary').innerHTML = `Итого: <b>${total} ₽</b>`;
        }

        function exportStatsToCSV() {
            const selected = document.getElementById('filterEmployee').value;
            const daysWorked = {}; Object.keys(shifts).forEach(date => { const list = selected==='all' ? shifts[date] : shifts[date].filter(s=>s.employee===selected); list.forEach(s => { daysWorked[s.employee] = (daysWorked[s.employee]||0)+1; }); });
            const emps = selected==='all' ? employees : employees.filter(e=>e.name===selected);
            let csv="data:text/csv;charset=utf-8,\uFEFFСотрудник;Ставка;Дни;Сумма\n"; emps.forEach(e => { const days=daysWorked[e.name]||0; const sum=days*e.rate; csv+=`"${e.name}";${e.rate};${days};${sum}\n`; });
            const link=document.createElement('a'); link.href=encodeURI(csv); link.download=(selected==='all'?'itogi.csv':`${selected}.csv`); document.body.appendChild(link); link.click(); document.body.removeChild(link);
        }

        document.addEventListener('DOMContentLoaded', () => { renderEmployeeList(); initSelect(); populateFilterSelect(); renderCalendar(); recalcStats(); });
    </script>
